cmake_minimum_required(VERSION 3.16)
project(serialized_proof_adapter LANGUAGES CXX)
find_package(SEAL CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(serialized_proof_bench driver.cpp vendor/src/client.cpp vendor/src/server.cpp vendor/src/pirparams.cpp)
target_include_directories(serialized_proof_bench PRIVATE vendor/header vendor/src vendor)
target_compile_features(serialized_proof_bench PRIVATE cxx_std_17)
target_compile_options(serialized_proof_bench PRIVATE -O3)
target_link_libraries(serialized_proof_bench PRIVATE SEAL::seal OpenSSL::Crypto)
